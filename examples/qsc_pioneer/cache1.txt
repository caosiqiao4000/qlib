    Integer calculateBarRishKByGrowth(Numeric psMultiNum,Numeric mainRenuveGrwoth)
    {
        If(psMultiNum <= 0)
        {
            Return 100;
        }
        Numeric rate = mainRenuveGrwoth / 100;
        
        Numeric current = 1;
        Numeric increatingNum = 1 * (1 + rate);
        
        Numeric years = (psMultiNum - current)/increatingNum;
        Return years;
    }
